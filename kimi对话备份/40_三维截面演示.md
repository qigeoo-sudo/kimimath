function updateSection3D() {
  cutPlane.position.y = zVal; // Three.js y = 数学z
  
  sectionLines.forEach(l => scene.remove(l));
  sectionLines = [];
  
  const steps = 600;
  const dx = (xMax - xMin) / steps;
  
  const intervals = [];
  let currentStart = null;
  
  for (let i = 0; i <= steps; i++) {
    const x = xMin + i * dx;
    const fx = a * x * x * x + b * x - zVal;
    if (fx >= 0) {
      if (currentStart === null) currentStart = i;
    } else {
      if (currentStart !== null) {
        intervals.push([currentStart, i - 1]);
        currentStart = null;
      }
    }
  }
  if (currentStart !== null) intervals.push([currentStart, steps]);
  
  intervals.forEach(([start, end]) => {
    const ptsUpper = [];
    const ptsLower = [];
    for (let i = start; i <= end; i++) {
      const x = xMin + i * dx;
      const fx = a * x * x * x + b * x - zVal;
      const y = Math.sqrt(fx); // 数学y
      // Three.js坐标：(tx, ty, tz) = (数学x, 数学z, 数学y)
      ptsUpper.push(new THREE.Vector3(x, zVal, y));
      ptsLower.push(new THREE.Vector3(x, zVal, -y));
    }
    if (ptsUpper.length > 1) {
      const geo = new THREE.BufferGeometry().setFromPoints(ptsUpper);
      const line = new THREE.Line(geo, sectionMat);
      scene.add(line);
      sectionLines.push(line);
    }
    if (ptsLower.length > 1) {
      const geo = new THREE.BufferGeometry().setFromPoints(ptsLower);
      const line = new THREE.Line(geo, sectionMat);
      scene.add(line);
      sectionLines.push(line);
    }
  });
}

让我逐一处理：**问题2：截面曲线没有封闭**
在 3D 中，截面曲线由多个区间组成。每个区间 [x_start, x_end] 对应 fx >= 0。在区间的端点处 fx = 0，所以 y = 0。上半支从 x_start 到 x_end，下半支从 x_end 回到 x_start。如果采样恰好包含端点，那么上半支的终点 (x_end, 0) 和下半支的起点 (x_end, 0) 是同一个点，下半支的终点 (x_start, 0) 和上半支的起点 (x_start, 0) 也是同一个点，这样就是一个闭合环。但问题是：当 fx 在某些 x 处从正变负时，我们的采样可能恰好跳过了 fx=0 的点。例如，如果根不在采样网格上，那么上一个采样点 fx>0，下一个采样点 fx<0，我们只取 fx>=0 的点，所以区间的端点不是精确的根，y 值也不是 0。这样上半支的终点和下半支的起点就不连接，形成开口。修复方案：在每个区间的端点处，如果端点不是精确的根位置，我们需要额外添加根处的点 (root, 0) 来确保闭合。或者更简单：在绘制每个区间时，确保在区间端点处 y=0。实际上，更好的方法是：对于每个区间，找到精确的左右端点（根），然后在绘制上半支和下半支时，确保端点包含 y=0。让我重新思考 3D 截面线的生成：当前代码：

function getIntervals(aa, bb, zz) {
  const intervals = [];
  const steps = 2000;
  const dxx = (xMax - xMin) / steps;
  let prev = aa * xMin * xMin * xMin + bb * xMin - zz;
  let currentLeft = null;
  
  for (let i = 1; i <= steps; i++) {
    const x = xMin + i * dxx;
    const curr = aa * x * x * x + bb * x - zz;
    
    // 从左根进入正区间 (prev <= 0, curr > 0)
    if (prev <= 0 && curr > 0) {
      let lo = x - dxx, hi = x;
      for (let k = 0; k < 25; k++) {
        const mid = (lo + hi) / 2;
        const vm = aa * mid * mid * mid + bb * mid - zz;
        if (Math.abs(vm) < 1e-12) { lo = hi = mid; break; }
        if (vm <= 0) lo = mid; else hi = mid;
      }
      currentLeft = (lo + hi) / 2;
    }
    
    // 从正区间出去 (prev > 0, curr <= 0)
    if (prev > 0 && curr <= 0) {
      let lo = x - dxx, hi = x;
      for (let k = 0; k < 25; k++) {
        const mid = (lo + hi) / 2;
        const vm = aa * mid * mid * mid + bb * mid - zz;
        if (Math.abs(vm) < 1e-12) { lo = hi = mid; break; }
        if (vm > 0) lo = mid; else hi = mid;
      }
      const rightRoot = (lo + hi) / 2;
      if (currentLeft !== null) {
        intervals.push({ left: currentLeft, right: rightRoot });
        currentLeft = null;
      }
    }
    
    prev = curr;
  }
  
  // 处理左边界
  const fLeft = aa * xMin * xMin * xMin + bb * xMin - zz;
  if (fLeft > 0) {
    intervals.unshift({ left: xMin, right: currentLeft !== null ? currentLeft : xMax });
  } else if (currentLeft !== null) {
    intervals.push({ left: currentLeft, right: xMax });
  }
  
  return intervals;
}
