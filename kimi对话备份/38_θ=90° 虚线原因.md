采用从右到左的算法？不，用户明确要求从左到右。**最终方案**：初始 R=O （不显示在画布上，或者显示为一个特殊标记）。第一步：

- 倍点： 2O=O （切线就是无穷远直线，不画或画一条提示线）
- 若 bit=1： R=O+G=G （连线就是过 O 和 G 的直线，即过 G 的垂直线，交点就是 −G ，对称得 G ）

这太复杂了。对于教学演示，我建议：**简化**：假设私钥的二进制表示没有前导零（即第一位总是 1）。初始 R=G 。然后对**剩余的位**（从第二位到末位）执行：

- 若当前位为 1： R←R+G

这样对于 13 = 1101：

- 初始 R = G（对应第一个 1）
- bit=1（第二位）：R = 2G，然后 R = 2G+G = 3G
- bit=0（第三位）：R = 2(3G) = 6G
- bit=1（第四位）：R = 2(6G) = 12G，然后 R = 12G+G = 13G

```javascript
function ecDouble(P) {
  if (P.isInfinity) return { isInfinity: true };
  const { X, Y } = P;
  const lambda = (3 * X * X + A) / (2 * Y);
  const X3 = lambda * lambda - 2 * X;
  const Y3 = lambda * (X - X3) - Y;
  return { X: X3, Y: Y3 };
}

function ecAdd(P, Q) {
  if (P.isInfinity) return Q;
  if (Q.isInfinity) return P;
  const { X: X1, Y: Y1 } = P;
  const { X: X2, Y: Y2 } = Q;
  if (Math.abs(X1 - X2) < 1e-10 && Math.abs(Y1 + Y2) < 1e-10) {
    return { isInfinity: true }; // P + (-P) = O
  }
  const lambda = (Y2 - Y1) / (X2 - X1);
  const X3 = lambda * lambda - X1 - X2;
  const Y3 = lambda * (X1 - X3) - Y1;
  return { X: X3, Y: Y3 };
}
```

```javascript
function precomputeSteps(baseStd, n) {
  const bits = n.toString(2).split('');
  const steps = [];
  let R = { isInfinity: true };
  
  for (let i = 0; i < bits.length; i++) {
    // Double
    const R_double = ecDouble(R);
    steps.push({
      type: 'double',
      bitIndex: i,
      bit: bits[i],
      from: {...R},
      to: {...R_double},
      // 几何信息在标准坐标下计算
    });
    R = R_double;
    
    if (bits[i] === '1') {
      const R_add = ecAdd(R, baseStd);
      steps.push({
        type: 'add',
        bitIndex: i,
        bit: bits[i],
        from: {...R},
        to: {...R_add},
        base: {...baseStd}
      });
      R = R_add;
    }
  }
  
  return { bits, steps, result: R };
}
```
