- 用户在椭圆/抛物线状态下调整了参数 → 但如果是通过拖动 slider 离开该状态的，`savedState['ellipse']` 等**不会被更新**

function updateAlpha(deg) {
    const rad = deg * Math.PI / 180;
    // 保存旧状态（如果类型即将变化）
    const curType = getCurrentType();
    if (curType) {
        const testParams = computeDandelin(rad, currentGammaDeg * Math.PI / 180);
        const newType = testParams.isDegenerate ? 'degenerate' :
                        testParams.isCircle ? 'circle' :
                        testParams.isEllipse ? 'ellipse' :
                        testParams.isParabola ? 'parabola' :
                        testParams.isHyperbola ? 'hyperbola' : null;
        if (curType !== newType && savedState[curType]) {
            savedState[curType] = {
                alpha: currentSemiAngleDeg, gamma: currentGammaDeg, p: currentAngle,
                camX: camera.position.x, camY: camera.position.y, camZ: camera.position.z,
                tX: controls.target.x, tY: controls.target.y, tZ: controls.target.z
            };
        }
    }
    currentSemiAngleDeg = deg;
    document.getElementById('alpha-angle').textContent = deg.toFixed(2) + '°';
    const e_val = Math.cos(currentGammaDeg * Math.PI / 180) / Math.cos(rad);
    document.getElementById('tan-display').textContent = 'e=cosγ/cosα=' + e_val.toFixed(3);
    buildScene(rad, currentGammaDeg * Math.PI / 180);
    updateScene(currentAngle);
}

function updateGamma(deg) {

    const curType = getCurrentType();

            savedState[curType] = {
                alpha: currentSemiAngleDeg, gamma: currentGammaDeg, p: currentAngle,

                tX: controls.target.x, tY: controls.target.y, tZ: controls.target.z
            };
        }

    buildScene(currentSemiAngleDeg * Math.PI / 180, deg * Math.PI / 180);
    updateScene(currentAngle);
}
