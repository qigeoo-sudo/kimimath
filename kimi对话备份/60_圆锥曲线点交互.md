        function computeDandelin(alphaRad, gammaRad) {
            const H = CONE_H;
            const h0 = INTERCEPT;
            const sinA = Math.sin(alphaRad);
            const cosA = Math.cos(alphaRad);
            const betaRad = Math.PI / 2 - gammaRad;
            const sinB = Math.sin(betaRad);
            const cosB = Math.cos(betaRad);
            const nUnit = new THREE.Vector3(sinB, -cosB, 0).normalize();
            const P0 = new THREE.Vector3(0, h0, 0);

            const isEllipse = gammaRad > alphaRad + 1e-6;
            const isParabola = Math.abs(gammaRad - alphaRad) <= 1e-6;
            const isHyperbola = gammaRad < alphaRad - 1e-6;
            const isDegenerate = gammaRad < 1e-3;

            const y1 = (H * sinA + h0 * cosB) / (sinA + cosB);
            const R1 = (y1 - h0) * cosB;
            const O1 = new THREE.Vector3(0, y1, 0);
            const d1 = (O1.x - P0.x) * nUnit.x + (O1.y - P0.y) * nUnit.y + (O1.z - P0.z) * nUnit.z;
            const F1 = O1.clone().sub(nUnit.clone().multiplyScalar(d1));
            const y_t1 = H * sinA * sinA + y1 * cosA * cosA;
            const r1_circle = (H - y_t1) * Math.tan(alphaRad);

            let y2, R2, O2, F2, y_t2, r2_circle;
            let lowerValid = true;
            if (isParabola) {
                lowerValid = false;
                y2 = 0; R2 = 0; O2 = new THREE.Vector3(0,0,0); F2 = new THREE.Vector3(0,0,0);
                y_t2 = 0; r2_circle = 0;
            } else if (isHyperbola) {
                const denom = cosB - sinA;
                y2 = (h0 * cosB - H * sinA) / denom;
                R2 = (y2 - H) * sinA;
                if (R2 < 0.01 || y2 <= H) { lowerValid = false; y2 = H; R2 = 0; }
                O2 = new THREE.Vector3(0, y2, 0);
                const d2 = (O2.x - P0.x) * nUnit.x + (O2.y - P0.y) * nUnit.y + (O2.z - P0.z) * nUnit.z;
                F2 = O2.clone().sub(nUnit.clone().multiplyScalar(d2));
                y_t2 = H * sinA * sinA + y2 * cosA * cosA;
                r2_circle = (y_t2 - H) * Math.tan(alphaRad);
            } else {
                const denom = sinA - cosB;
                y2 = (H * sinA - h0 * cosB) / denom;
                R2 = (h0 - y2) * cosB;
                if (R2 <= 0.01) { lowerValid = false; y2 = 0; R2 = 0; }
                O2 = new THREE.Vector3(0, y2, 0);
                const d2 = (O2.x - P0.x) * nUnit.x + (O2.y - P0.y) * nUnit.y + (O2.z - P0.z) * nUnit.z;
                F2 = O2.clone().sub(nUnit.clone().multiplyScalar(d2));
                y_t2 = y2 + R2 * sinA;
                r2_circle = (H - y_t2) * Math.tan(alphaRad);
            }

            let center, aLen, bLen, uDir, vDir;
            let directrixPoint = null, directrixDir = null, vertex = null, focalParam = null;
            let isCircle = false;
            if (isEllipse) {
                const F1F2 = new THREE.Vector3().subVectors(F2, F1);
                const c = F1F2.length() / 2;
                isCircle = c < 1e-6;
                const a = Math.abs(y_t1 - y_t2) / (2 * cosA);
                const b = Math.sqrt(Math.max(0, a * a - c * c));
                center = new THREE.Vector3().addVectors(F1, F2).multiplyScalar(0.5);
                if (F1F2.length() < 1e-6) {
                    uDir = new THREE.Vector3(1, 0, 0);
                } else {
                    uDir = F1F2.clone().normalize();
                }
                vDir = new THREE.Vector3().crossVectors(nUnit, uDir).normalize();
                aLen = a; bLen = b;
            } else if (isHyperbola) {
                const F1F2 = new THREE.Vector3().subVectors(F1, F2);
                const c = F1F2.length() / 2;
                const a = Math.abs(y_t1 - y_t2) / (2 * cosA);
                const b = Math.sqrt(Math.max(0, c * c - a * a));
                center = new THREE.Vector3().addVectors(F1, F2).multiplyScalar(0.5);
                uDir = F1F2.clone().normalize();
                vDir = new THREE.Vector3().crossVectors(nUnit, uDir).normalize();
                aLen = a; bLen = b;

                vertex = F1.clone().addScaledVector(vDir, -focalParam/2);
                center = vertex.clone();
                aLen = 0; bLen = 0;
            }

            function computeDirectrixForLevel(y_t, n, h0_val) {
                const dir = new THREE.Vector3().crossVectors(n, new THREE.Vector3(0, 1, 0)).normalize();
                if (dir.length() < 0.001) dir.set(1, 0, 0);
                let pt;
                if (Math.abs(n.z) > 0.001) {
                    const z = -n.y * (y_t - h0_val) / n.z;
                    pt = new THREE.Vector3(0, y_t, z);

                    const x = -n.y * (y_t - h0_val) / n.x;
                    pt = new THREE.Vector3(x, y_t, 0);
                } else {
                    pt = new THREE.Vector3(0, y_t, 0);
                }
                return { point: pt, dir: dir };
            }

                directrixPoint, directrixDir, vertex, focalParam,
                directrixData,   // 新增：准线数据数组
                isDegenerate,
                isCircle,
                alphaRad
            };
        }

        function getConicPoint(params, t, branch = 0) {
            if (params.isDegenerate) {
                const sign = (branch >= 0) ? 1 : -1;
                const cosA = Math.cos(params.alphaRad);
                const sinA = Math.sin(params.alphaRad);
                const dir = new THREE.Vector3(0, -cosA, sign * sinA);
                const apex = new THREE.Vector3(0, params.coneHeight, 0);
                return apex.clone().addScaledVector(dir, t);
            }
            if (params.isEllipse) {
                const cos_t = Math.cos(t), sin_t = Math.sin(t);
                return new THREE.Vector3(
                    params.center.x + params.aLen * cos_t * params.uDir.x + params.bLen * sin_t * params.vDir.x,
                    params.center.y + params.aLen * cos_t * params.uDir.y + params.bLen * sin_t * params.vDir.y,
                    params.center.z + params.aLen * cos_t * params.uDir.z + params.bLen * sin_t * params.vDir.z
                );
            } else if (params.isHyperbola) {
                const cosh_t = Math.cosh(t), sinh_t = Math.sinh(t);
                const sign = (branch >= 0) ? 1 : -1;
                return new THREE.Vector3(
                    params.center.x + sign * params.aLen * cosh_t * params.uDir.x + params.bLen * sinh_t * params.vDir.x,
                    params.center.y + sign * params.aLen * cosh_t * params.uDir.y + params.bLen * sinh_t * params.vDir.y,
                    params.center.z + sign * params.aLen * cosh_t * params.uDir.z + params.bLen * sinh_t * params.vDir.z
                );
            } else if (params.isParabola) {
                const p = Math.max(params.focalParam, 0.1);
                return params.vertex.clone()
                    .addScaledVector(params.uDir, t)
                    .addScaledVector(params.vDir, (t * t) / (2 * p));
            }
            return new THREE.Vector3();
        }

        function getGeneratorIntersections(params, P) {

                B = apex.clone().add(dir.clone().multiplyScalar(s2));
            } else B = apex.clone();
            return { A, B, dir };
        }

        function computeDandelin(alphaRad, gammaRad) {
            const H = CONE_H;
            const h0 = INTERCEPT;
            const sinA = Math.sin(alphaRad);
            const cosA = Math.cos(alphaRad);
            const betaRad = Math.PI / 2 - gammaRad;
            const sinB = Math.sin(betaRad);
            const cosB = Math.cos(betaRad);
            const nUnit = new THREE.Vector3(sinB, -cosB, 0).normalize();
            const P0 = new THREE.Vector3(0, h0, 0);

            const isEllipse = gammaRad > alphaRad + 1e-6;
            const isParabola = Math.abs(gammaRad - alphaRad) <= 1e-6;
            const isHyperbola = gammaRad < alphaRad - 1e-6;
            const isDegenerate = gammaRad < 1e-3;

            const y1 = (H * sinA + h0 * cosB) / (sinA + cosB);
            const R1 = (y1 - h0) * cosB;
            const O1 = new THREE.Vector3(0, y1, 0);
            const d1 = (O1.x - P0.x) * nUnit.x + (O1.y - P0.y) * nUnit.y + (O1.z - P0.z) * nUnit.z;
            const F1 = O1.clone().sub(nUnit.clone().multiplyScalar(d1));
            const y_t1 = H * sinA * sinA + y1 * cosA * cosA;
            const r1_circle = (H - y_t1) * Math.tan(alphaRad);

            let y2, R2, O2, F2, y_t2, r2_circle;
            let lowerValid = true;
            if (isParabola) {
                lowerValid = false;
                y2 = 0; R2 = 0; O2 = new THREE.Vector3(0,0,0); F2 = new THREE.Vector3(0,0,0);
                y_t2 = 0; r2_circle = 0;
            } else if (isHyperbola) {
                const denom = cosB - sinA;
                y2 = (h0 * cosB - H * sinA) / denom;
                R2 = (y2 - H) * sinA;
                if (R2 < 0.01 || y2 <= H) { lowerValid = false; y2 = H; R2 = 0; }
                O2 = new THREE.Vector3(0, y2, 0);
                const d2 = (O2.x - P0.x) * nUnit.x + (O2.y - P0.y) * nUnit.y + (O2.z - P0.z) * nUnit.z;
                F2 = O2.clone().sub(nUnit.clone().multiplyScalar(d2));
                y_t2 = H * sinA * sinA + y2 * cosA * cosA;
                r2_circle = (y_t2 - H) * Math.tan(alphaRad);
            } else {
                const denom = sinA - cosB;
                y2 = (H * sinA - h0 * cosB) / denom;
                R2 = (h0 - y2) * cosB;
                if (R2 <= 0.01) { lowerValid = false; y2 = 0; R2 = 0; }
                O2 = new THREE.Vector3(0, y2, 0);
                const d2 = (O2.x - P0.x) * nUnit.x + (O2.y - P0.y) * nUnit.y + (O2.z - P0.z) * nUnit.z;
                F2 = O2.clone().sub(nUnit.clone().multiplyScalar(d2));
                y_t2 = y2 + R2 * sinA;
                r2_circle = (H - y_t2) * Math.tan(alphaRad);
            }

            let center, aLen, bLen, uDir, vDir;
            let directrixPoint = null, directrixDir = null, vertex = null, focalParam = null;
            let isCircle = false;
            if (isEllipse) {
                const F1F2 = new THREE.Vector3().subVectors(F2, F1);
                const c = F1F2.length() / 2;
                isCircle = c < 1e-6;
                const a = Math.abs(y_t1 - y_t2) / (2 * cosA);
                const b = Math.sqrt(Math.max(0, a * a - c * c));
                center = new THREE.Vector3().addVectors(F1, F2).multiplyScalar(0.5);

                } else {

            } else if (isHyperbola) {
                const F1F2 = new THREE.Vector3().subVectors(F1, F2);
                const c = F1F2.length() / 2;
                const a = Math.abs(y_t1 - y_t2) / (2 * cosA);
                const b = Math.sqrt(Math.max(0, c * c - a * a));

                uDir = F1F2.clone().normalize();
                vDir = new THREE.Vector3().crossVectors(nUnit, uDir).normalize();
                aLen = a; bLen = b;

                center = vertex.clone();

            function computeDirectrixForLevel(y_t, n, h0_val) {
                const dir = new THREE.Vector3().crossVectors(n, new THREE.Vector3(0, 1, 0)).normalize();
                if (dir.length() < 0.001) dir.set(1, 0, 0);
                let pt;
                if (Math.abs(n.z) > 0.001) {

                    pt = new THREE.Vector3(x, y_t, 0);
                } else {
                    pt = new THREE.Vector3(0, y_t, 0);
                }
                return { point: pt, dir: dir };

                isCircle,
                alphaRad
            };
        }

        function getConicPoint(params, t, branch = 0) {
            if (params.isDegenerate) {
                const sign = (branch >= 0) ? 1 : -1;
                const cosA = Math.cos(params.alphaRad);
                const sinA = Math.sin(params.alphaRad);
                const dir = new THREE.Vector3(0, -cosA, sign * sinA);
                const apex = new THREE.Vector3(0, params.coneHeight, 0);
                return apex.clone().addScaledVector(dir, t);
            }
            if (params.isEllipse) {
                const cos_t = Math.cos(t), sin_t = Math.sin(t);
                return new THREE.Vector3(
                    params.center.x + params.aLen * cos_t * params.uDir.x + params.bLen * sin_t * params.vDir.x,
                    params.center.y + params.aLen * cos_t * params.uDir.y + params.bLen * sin_t * params.vDir.y,
                    params.center.z + params.aLen * cos_t * params.uDir.z + params.bLen * sin_t * params.vDir.z
                );
            } else if (params.isHyperbola) {
                const cosh_t = Math.cosh(t), sinh_t = Math.sinh(t);
                const sign = (branch >= 0) ? 1 : -1;
                return new THREE.Vector3(
                    params.center.x + sign * params.aLen * cosh_t * params.uDir.x + params.bLen * sinh_t * params.vDir.x,
                    params.center.y + sign * params.aLen * cosh_t * params.uDir.y + params.bLen * sinh_t * params.vDir.y,
                    params.center.z + sign * params.aLen * cosh_t * params.uDir.z + params.bLen * sinh_t * params.vDir.z
                );
            } else if (params.isParabola) {
                const p = Math.max(params.focalParam, 0.1);
                return params.vertex.clone()
                    .addScaledVector(params.uDir, t)
                    .addScaledVector(params.vDir, (t * t) / (2 * p));
            }
            return new THREE.Vector3();
        }

            const apex = new THREE.Vector3(0, params.coneHeight, 0);
            const dir = new THREE.Vector3().subVectors(P, apex).normalize();

                B = apex.clone().add(dir.clone().multiplyScalar(s2));
            } else B = apex.clone();
            return { A, B, dir };
        }

        let currentParams = null;
        let currentAngle = 0;

        function updateGeometry(params) {
            currentParams = params;
            const quat = new THREE.Quaternion().setFromUnitVectors(new THREE.Vector3(0,0,1), params.planeNormal);
            planeMesh.geometry = new THREE.PlaneGeometry(24, 24);
            planeMesh.geometry.applyQuaternion(quat);
            planeMesh.geometry.translate(0, INTERCEPT, 0);

            const pts = [];
            if (params.isDegenerate) {
                const range = params.coneHeight / Math.cos(params.alphaRad);
                const apex = new THREE.Vector3(0, params.coneHeight, 0);
                const cosA = Math.cos(params.alphaRad);
                const sinA = Math.sin(params.alphaRad);
                // 直线1：正圆锥 z>0 向下 ↔ 对顶锥 z<0 向上（共线）
                const down1 = apex.clone().add(new THREE.Vector3(0, -cosA * range, sinA * range));
                const up1 = apex.clone().add(new THREE.Vector3(0, cosA * range, -sinA * range));
                conicLine.geometry.setFromPoints([down1, apex, up1]);
                // 直线2：正圆锥 z<0 向下 ↔ 对顶锥 z>0 向上（共线）
                const down2 = apex.clone().add(new THREE.Vector3(0, -cosA * range, -sinA * range));
                const up2 = apex.clone().add(new THREE.Vector3(0, cosA * range, sinA * range));
                conicLine2.geometry.setFromPoints([down2, apex, up2]);
                conicLine2.visible = true;
            } else if (params.isEllipse) {
                for (let i = 0; i <= 200; i++) pts.push(getConicPoint(params, (i/200)*Math.PI*2));
            } else if (params.isHyperbola) {
                const range = 3.0, steps = 120;
                const pts1 = [], pts2 = [];
                for (let i = 0; i <= steps; i++) {
                    const t = -range + (i/steps)*range*2;
                    pts1.push(getConicPoint(params, t, 1));
                }
                for (let i = 0; i <= steps; i++) {
                    const t = -range + (i/steps)*range*2;
                    pts2.push(getConicPoint(params, t, -1));
                }
                conicLine.geometry.setFromPoints(pts1);
                conicLine2.geometry.setFromPoints(pts2);
            } else if (params.isParabola) {
                for (let i = 0; i <= 200; i++) {
                    const t = -8 + (i/200)*16;
                    pts.push(getConicPoint(params, t));
                }
            }
            if (params.isHyperbola) {
                conicLine2.visible = true;
            } else {
                conicLine2.visible = false;
                conicLine.geometry.setFromPoints(pts);
            }

            // 更新六等分点位置
            if (hexPoints.length === 6) {
                for (let i = 0; i < 6; i++) {
                    let Pt;
                    if (params.isDegenerate) {
                        const cosA = Math.cos(params.alphaRad), sinA = Math.sin(params.alphaRad);
                        const apex = new THREE.Vector3(0, params.coneHeight, 0);
                        const range = params.coneHeight / cosA;
                        const ang = hexAngles[i];
                        if (ang <= 180) {
                            const frac = ang / 180, s = frac * (2 * range) - range;
                            Pt = s < 0 ? apex.clone().addScaledVector(new THREE.Vector3(0, -cosA, sinA), -s)
                                       : apex.clone().addScaledVector(new THREE.Vector3(0, cosA, -sinA), s);
                        } else {
                            const frac = (ang - 180) / 180, s = frac * (2 * range) - range;
                            Pt = s < 0 ? apex.clone().addScaledVector(new THREE.Vector3(0, -cosA, -sinA), -s)
                                       : apex.clone().addScaledVector(new THREE.Vector3(0, cosA, sinA), s);
                        }
                    } else if (params.isEllipse) {
                        Pt = getConicPoint(params, hexAngles[i] * Math.PI / 180);
                    } else if (params.isHyperbola) {
                        const range = 2.5, br = (hexAngles[i] <= 180) ? -1 : 1;
                        const t = -range + ((hexAngles[i] % 180) / 180) * range * 2;
                        Pt = getConicPoint(params, t, br);
                    } else if (params.isParabola) {
                        Pt = getConicPoint(params, (hexAngles[i] - 180) * 0.05);
                    }
                    if (Pt) { hexPoints[i].position.copy(Pt); hexLabels[i].position.copy(Pt); }
                }
            }

            // 抛物线：在切圆水平面上创建正方形（中心为切圆中心，边长=3×切圆直径）
            if (params.isParabola && paraSquare) {
                const sqHalf = params.r1 * 3; // 边长的一半 = 3 × r1（直径的1.5倍，即直径的3倍的一半）
                const sqY = params.y_t1;
                const sqGeo = new THREE.BufferGeometry();
                const sqVerts = new Float32Array([
                    -sqHalf, sqY, -sqHalf,   sqHalf, sqY, -sqHalf,
                     sqHalf, sqY,  sqHalf,  -sqHalf, sqY,  sqHalf
                ]);
                const sqIndices = [0, 1, 2, 0, 2, 3];
                sqGeo.setAttribute('position', new THREE.BufferAttribute(sqVerts, 3));
                sqGeo.setIndex(sqIndices);
                sqGeo.computeVertexNormals();
                paraSquare.geometry = sqGeo;
                paraSquare.visible = true;
            } else if (paraSquare) {
                paraSquare.visible = false;
            }

            const showLower = params.lowerValid && !params.isDegenerate;
            const showUpper = !params.isDegenerate;
            sphere1.visible = showUpper; sphereEdges1.visible = showUpper;
            if (showUpper) {
                sphere1.position.copy(params.O1);
                sphere1.geometry = new THREE.SphereGeometry(Math.max(params.R1, 0.01), 32, 32);
                sphereEdges1.position.copy(params.O1);
                sphereEdges1.geometry = new THREE.EdgesGeometry(new THREE.SphereGeometry(Math.max(params.R1, 0.01), 32, 32));
            }
            sphere2.visible = showLower; sphereEdges2.visible = showLower;
            if (showLower) {
                sphere2.position.copy(params.O2);
                sphere2.geometry = new THREE.SphereGeometry(Math.max(params.R2, 0.01), 32, 32);
                sphereEdges2.position.copy(params.O2);
                sphereEdges2.geometry = new THREE.EdgesGeometry(new THREE.SphereGeometry(Math.max(params.R2, 0.01), 32, 32));
            }

            circle1.visible = showUpper;
            circle2.visible = showLower;
            if (showUpper) {
                circle1.geometry.setFromPoints(Array.from({ length: 65 }, (_, i) => {
                    const a = (i/64)*Math.PI*2;
                    return new THREE.Vector3(params.r1 * Math.cos(a), params.y_t1, params.r1 * Math.sin(a));
                }));
            }
            if (showLower) {

function angleFromScreen(clientX, clientY, idx) {
    const rect = renderer.domElement.getBoundingClientRect();
    const mx = ((clientX - rect.left) / rect.width) * 2 - 1;
    const my = -((clientY - rect.top) / rect.height) * 2 + 1;
    const ray = new THREE.Raycaster();

            const dx = target.x - currentParams.center.x;
            const dy = target.y - currentParams.center.y;
            const dz = target.z - currentParams.center.z;
            const u = dx * currentParams.uDir.x + dy * currentParams.uDir.y + dz * currentParams.uDir.z;
