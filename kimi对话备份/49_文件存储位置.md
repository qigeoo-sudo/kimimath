            const dx2 = p4.x - p3.x, dy2 = p4.y - p3.y, dz2 = p4.z - p3.z;

            mesh.position.copy(position);
            return mesh;
        }

        function createTriangle(p1, p2, p3, color, opacity) {
            const geometry = new THREE.BufferGeometry();
            const vertices = new Float32Array([
                p1.x, p1.y, p1.z,
                p2.x, p2.y, p2.z,
                p3.x, p3.y, p3.z
            ]);
            geometry.setAttribute('position', new THREE.BufferAttribute(vertices, 3));
            geometry.computeVertexNormals();
            const material = getMat(color, true, opacity, false);
            return new THREE.Mesh(geometry, material);
        }

        function createPlaneMesh(normal, point, size = 15, color = 0x888888, opacity = 0.15) {
            const planeGeometry = new THREE.PlaneGeometry(size * 2, size * 2);
            const planeMaterial = new THREE.MeshPhongMaterial({
                color: color,
                transparent: true,
                opacity: opacity,
                side: THREE.DoubleSide,
                depthWrite: false
            });
            const mesh = new THREE.Mesh(planeGeometry, planeMaterial);

            // 对齐平面：PlaneGeometry 默认法向为 +Z，用 lookAt 使其指向目标法向
            mesh.position.copy(point);
            mesh.lookAt(point.clone().add(normal));
            return mesh;
        }

        const gridHelper = new THREE.GridHelper(20, 20, 0x1a1a2e, 0x1a1a2e);
        gridHelper.position.y = 0;
        scene.add(gridHelper);

        axesGroup.add(createLine(new THREE.Vector3(0,0,0), new THREE.Vector3(0,2,0), 0x00ff00));
        axesGroup.add(createLine(new THREE.Vector3(0,0,0), new THREE.Vector3(0,0,2), 0x0000ff));
        scene.add(axesGroup);

        // ==================== 获取平面参数 ====================
        function getPlane() {
            const tx = parseFloat(document.getElementById('tiltX').value) * Math.PI / 180;
            const ty = parseFloat(document.getElementById('tiltY').value) * Math.PI / 180;
            const tz = parseFloat(document.getElementById('tiltZ').value) * Math.PI / 180;
            const pz = parseFloat(document.getElementById('posZ').value);
            let n = new THREE.Vector3(0, 1, 0);
            n = rotateX(n, tx);
            n = rotateY(n, ty);
            n = rotateZ(n, tz);
            return { normal: Vec3.normalize(n), point: new THREE.Vector3(0, 2 + pz, 0) };
        }

        function updateScene() {

                if (child.geometry) child.geometry.dispose();
            }
            clearLabels();

            const plane = getPlane();

            // 计算截面点
            const secPts = [];
            [[S, A], [S, B], [S, C]].forEach(([s, e]) => {
                const ip = linePlaneIntersect(s, e, plane.normal, plane.point);
                if (ip) secPts.push(ip);
            });

            if (secPts.length < 3) return;

            const Ap = secPts[0], Bp = secPts[1], Cp = secPts[2];

            // 底面三角形
            if (!hideOthersMode || !projectionMode) {
                mainGroup.add(createTriangle(A, B, C, 0xff4444, 0.12));
                mainGroup.add(createLine(A, B, 0xff4444, 1));
                mainGroup.add(createLine(B, C, 0xff4444, 1));
                mainGroup.add(createLine(C, A, 0xff4444, 1));

                mainGroup.add(createLine(S, B, 0x666666, 1));

                mainGroup.add(createLine(Bp, Cp, 0xffcc00, 1));

                // 顶点
                mainGroup.add(createPointMesh(S, 0xff66ff, 0.12));
                mainGroup.add(createPointMesh(A, 0xff4444, 0.08));
                mainGroup.add(createPointMesh(B, 0xff4444, 0.08));
                mainGroup.add(createPointMesh(C, 0xff4444, 0.08));
                mainGroup.add(createPointMesh(Ap, 0xffcc00, 0.08));
                mainGroup.add(createPointMesh(Bp, 0xffcc00, 0.08));
                mainGroup.add(createPointMesh(Cp, 0xffcc00, 0.08));

                addLabel(createLabel('S', S, '#ff66ff'), mainGroup);
                addLabel(createLabel('A', A, '#ff4444'), mainGroup);
                addLabel(createLabel('B', B, '#ff4444'), mainGroup);
                addLabel(createLabel('C', C, '#ff4444'), mainGroup);
                addLabel(createLabel("A'", Ap, '#ffcc00'), mainGroup);
                addLabel(createLabel("B'", Bp, '#ffcc00'), mainGroup);
                addLabel(createLabel("C'", Cp, '#ffcc00'), mainGroup);

                // 延长线
                function drawExtension(p1, p2, color) {
                    const dir = Vec3.normalize(Vec3.sub(p2, p1));
                    const far1 = Vec3.sub(p1, Vec3.mul(dir, 15));
                    const far2 = Vec3.add(p2, Vec3.mul(dir, 15));
                    const line = createLine(far1, far2, color, 0.6);
                    line.material.dashSize = 0.2;
                    line.material.gapSize = 0.1;
                    // Three.js LineBasicMaterial doesn't support dash natively easily, use opacity instead
                    line.material.opacity = 0.4;
                    mainGroup.add(line);
                }

                drawExtension(A, B, 0x00aaff);
                drawExtension(Ap, Bp, 0x00aaff);
                drawExtension(B, C, 0x00aaff);
                drawExtension(Bp, Cp, 0x00aaff);
                drawExtension(C, A, 0x00aaff);
                drawExtension(Cp, Ap, 0x00aaff);

                // 交点
                const P = lineIntersection3D(A, B, Ap, Bp);
                const Q = lineIntersection3D(B, C, Bp, Cp);
                const R = lineIntersection3D(C, A, Cp, Ap);

                if (P && Q && R) {
                    mainGroup.add(createPointMesh(P, 0x00ffcc, 0.1));
                    mainGroup.add(createPointMesh(Q, 0x00ffcc, 0.1));
                    mainGroup.add(createPointMesh(R, 0x00ffcc, 0.1));
                    addLabel(createLabel('P', P, '#00ffcc'), mainGroup);
                    addLabel(createLabel('Q', Q, '#00ffcc'), mainGroup);
                    addLabel(createLabel('R', R, '#00ffcc'), mainGroup);

                    // Desargues线
                    const dir = Vec3.normalize(Vec3.sub(Q, P));
                    const d1 = Vec3.sub(P, Vec3.mul(dir, 20));
                    const d2 = Vec3.add(P, Vec3.mul(dir, 20));
                    const desLine = createLine(d1, d2, 0x00ff88, 1);
                    desLine.material.linewidth = 2; // Note: linewidth only works on some browsers
                    desarguesLineGroup.add(desLine);

                    // 更新实时数据
                    document.getElementById('coord-P').textContent = `(${P.x.toFixed(2)},${P.y.toFixed(2)},${P.z.toFixed(2)})`;
                    document.getElementById('coord-Q').textContent = `(${Q.x.toFixed(2)},${Q.y.toFixed(2)},${Q.z.toFixed(2)})`;
                    document.getElementById('coord-R').textContent = `(${R.x.toFixed(2)},${R.y.toFixed(2)},${R.z.toFixed(2)})`;

                    const distR = pointToLineDistance(R, P, Q);
                    const maxC = Math.max(Math.abs(P.x), Math.abs(P.y), Math.abs(P.z), Math.abs(Q.x), Math.abs(Q.y), Math.abs(Q.z), Math.abs(R.x), Math.abs(R.y), Math.abs(R.z));
                    const relErr = maxC > 0.001 ? distR / maxC : distR;

                    const colEl = document.getElementById('collinearity');
                    if (relErr < 0.01) {
                        colEl.textContent = '<0.01';
                        colEl.style.color = '#4ade80';
                    } else {
                        colEl.textContent = relErr.toFixed(4);
                        colEl.style.color = '#f87171';
                    }
                }
            }

            // 中心投影模式
            if (projectionMode) {
                // Σ平面 - 奶白色不透明纸张效果
                const sigmaPlane = createPlaneMesh(new THREE.Vector3(0, 1, 0), new THREE.Vector3(0, sigmaY, 0), 15, 0xfaf8f5, 0.88);
                sigmaPlane.renderOrder = 0;
                projectionGroup.add(sigmaPlane);

                const points = [S, A, B, C, Ap, Bp, Cp];
                const labels_arr = ['S', 'A', 'B', 'C', "A'", "B'", "C'"];
                const projected = {};

                points.forEach((pt, i) => {
                    const proj = projectToSigma(pt, O, sigmaY);
                    if (!proj) return;
                    projected[labels_arr[i]] = proj;

                    // 投影线（hideOthersMode 时隐藏）
                    if (!hideOthersMode) {
                        const projLine = createLine(O, proj, 0x999999, 0.7);
                        projLine.renderOrder = 2;
                        projectionGroup.add(projLine);
                    }
                    // 投影点
                    projectionGroup.add(createPointMesh(proj, 0x000000, 0.08));
                    const projLbl = createLabel(labels_arr[i], proj, '#000000', 0.32);
                    projLbl.position.y += 0.18;
                    addLabel(projLbl, projectionGroup);
                });

                // 投影三角形边
                const projEdges = [
                    ['A', 'B'], ['B', 'C'], ['C', 'A'],
                    ["A'", "B'"], ["B'", "C'"], ["C'", "A'"]
                ];
                projEdges.forEach(([l1, l2]) => {
                    if (projected[l1] && projected[l2]) {
                        const isBase = ['A','B','C'].includes(l1) && ['A','B','C'].includes(l2);
                        const col = isBase ? 0x222222 : 0x111111;
                        const edgeLine = createLine(projected[l1], projected[l2], col, 1);
                        edgeLine.renderOrder = 2;
                        projectionGroup.add(edgeLine);
                    }
                });

                // 投影延长线
                const projExtPairs = [
                    ['A', 'B'], ["A'", "B'"],
                    ['B', 'C'], ["B'", "C'"],
                    ['C', 'A'], ["C'", "A'"]
                ];
                projExtPairs.forEach(([l1, l2]) => {
                    if (projected[l1] && projected[l2]) {
                        const dir = Vec3.normalize(Vec3.sub(projected[l2], projected[l1]));
                        const far1 = Vec3.sub(projected[l1], Vec3.mul(dir, 15));
                        const far2 = Vec3.add(projected[l2], Vec3.mul(dir, 15));
                        const isBase = ['A','B','C'].includes(l1) && ['A','B','C'].includes(l2);
                        const col = isBase ? 0x666666 : 0x555555;
                        const line = createLine(far1, far2, col, 0.6);
                        line.renderOrder = 2;
                        projectionGroup.add(line);
                    }
                });

                // 投影交点
                const projP = lineIntersection3D(projected['A'], projected['B'], projected["A'"], projected["B'"]);
                const projQ = lineIntersection3D(projected['B'], projected['C'], projected["B'"], projected["C'"]);
                const projR = lineIntersection3D(projected['C'], projected['A'], projected["C'"], projected["A'"]);

                if (projP && projQ && projR) {
                    projectionGroup.add(createPointMesh(projP, 0x000000, 0.1));
                    projectionGroup.add(createPointMesh(projQ, 0x000000, 0.1));
                    projectionGroup.add(createPointMesh(projR, 0x000000, 0.1));
                    const pLbl = createLabel('P', projP, '#000000', 0.32); pLbl.position.y += 0.18; addLabel(pLbl, projectionGroup);
                    const qLbl = createLabel('Q', projQ, '#000000', 0.32); qLbl.position.y += 0.18; addLabel(qLbl, projectionGroup);
                    const rLbl = createLabel('R', projR, '#000000', 0.32); rLbl.position.y += 0.18; addLabel(rLbl, projectionGroup);

                    const pDir = Vec3.normalize(Vec3.sub(projQ, projP));
                    const pd1 = Vec3.sub(projP, Vec3.mul(pDir, 15));
                    const pd2 = Vec3.add(projP, Vec3.mul(pDir, 15));

                    oLbl.position.y += 0.22;

                }
            }

        // 窗口大小调整
        window.addEventListener('resize', () => {
            camera.aspect = wrapper.clientWidth / wrapper.clientHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(wrapper.clientWidth, wrapper.clientHeight);

        // ==================== 动画循环 ====================
        let frameCounter = 0;
        let needsUpdate = true;

        function animate() {

            controls.update();
            frameCounter++;

            if (animating) {

                const period = Math.PI * 4;
                const t = animationTime;

                if (t >= period) {
                    // 演示完成：恢复原始状态

                    });

                } else {

                    document.getElementById('tiltX').value = animStartState.tiltX + 25 * Math.sin(t);

                    });
                    needsUpdate = true;
                }
            }

            // 每 10 帧或需要立即更新时才重建几何
            if (needsUpdate || frameCounter % 10 === 0) {
                updateScene();
                needsUpdate = false;
            }

            renderer.render(scene, camera);
        }

        // 初始化
        sliders.forEach(id => document.getElementById(id).dispatchEvent(new Event('input')));
        updateScene();
        animate();
    </script>
