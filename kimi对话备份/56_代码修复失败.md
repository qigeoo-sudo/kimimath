        function getConicPoint(params, t, branch = 0) {
            if (params.isDegenerate) {
                const sign = (branch >= 0) ? 1 : -1;
                const cosA = Math.cos(params.alphaRad);
                const sinA = Math.sin(params.alphaRad);
                const dir = new THREE.Vector3(0, -cosA, sign * sinA);
                const apex = new THREE.Vector3(0, params.coneHeight, 0);
                return apex.clone().addScaledVector(dir, t);
            }
            if (params.isEllipse) {
                const cos_t = Math.cos(t), sin_t = Math.sin(t);
                return new THREE.Vector3(
                    params.center.x + params.aLen * cos_t * params.uDir.x + params.bLen * sin_t * params.vDir.x,
                    params.center.y + params.aLen * cos_t * params.uDir.y + params.bLen * sin_t * params.vDir.y,
                    params.center.z + params.aLen * cos_t * params.uDir.z + params.bLen * sin_t * params.vDir.z
                );
            } else if (params.isHyperbola) {
                const cosh_t = Math.cosh(t), sinh_t = Math.sinh(t);
                const sign = (branch >= 0) ? 1 : -1;
                return new THREE.Vector3(
                    params.center.x + sign * params.aLen * cosh_t * params.uDir.x + params.bLen * sinh_t * params.vDir.x,
                    params.center.y + sign * params.aLen * cosh_t * params.uDir.y + params.bLen * sinh_t * params.vDir.y,
                    params.center.z + sign * params.aLen * cosh_t * params.uDir.z + params.bLen * sinh_t * params.vDir.z
                );
            } else if (params.isParabola) {
                const p = Math.max(params.focalParam, 0.1);
                return params.vertex.clone()
                    .addScaledVector(params.uDir, t)
                    .addScaledVector(params.vDir, (t * t) / (2 * p));
            }
            return new THREE.Vector3();
        }

        // ---- 新增：将双曲线的六等分点角度调整到红球分支（s > 0）----
        function adjustHexGeneratorsToRedBranch(params) {
            if (!params.isHyperbola) return;
            const H = params.coneHeight;
            const h0 = INTERCEPT;
            const sinA = Math.sin(params.alphaRad);
            const cosA = Math.cos(params.alphaRad);
            const sinG = Math.cos(params.alphaRad); // gamma 相关，用 planeNormal 推导
            // planeNormal = (cosγ, -sinγ, 0) (已归一化)
            // 从 planeNormal 反推 sinγ, cosγ
            const ny = params.planeNormal.y;
            const nx = params.planeNormal.x;
            // nUnit = (cosγ, -sinγ, 0) / norm
            // 由于 norm = sqrt(cos²γ + sin²γ) = 1，所以 nx = cosγ, ny = -sinγ
            const cosG = nx;
            const sinG_val = -ny;

            // 阈值: cosθ > -cotα·tanγ = -cosα·sinγ / (sinα·cosγ)
            const threshold = -cosA * sinG_val / (sinA * cosG);
            const clampedTh = Math.max(-1, Math.min(1, threshold));
            const thetaTh = Math.acos(clampedTh); // 边界弧度
            const margin = 3 * Math.PI / 180; // 3度余量

            // 安全边界（弧度）
            const bound1 = thetaTh - margin; // 正方向边界
            const bound2 = 2 * Math.PI - thetaTh + margin; // 负方向边界（即 360 - theta_th + margin）

            for (let i = 0; i < 6; i++) {
                let theta = hexGenerators[i];
                // 计算 s 符号
                const dir = new THREE.Vector3(
                    Math.cos(theta) * sinA,
                    -cosA,
                    Math.sin(theta) * sinA
                );
                const P0 = new THREE.Vector3(0, h0, 0);
                const apex = new THREE.Vector3(0, H, 0);
                const dirDotN = dir.dot(params.planeNormal);
                if (Math.abs(dirDotN) < 1e-10) continue;
                const s = P0.clone().sub(apex).dot(params.planeNormal) / dirDotN;

                if (s > 0) continue; // 已在红球分支

                // 需要调整角度到 s > 0 区域
                // s > 0 区域: (-thetaTh, thetaTh) mod 2π
                // 即 [0, thetaTh) ∪ (2π - thetaTh, 2π)
                // 安全区域: [margin, thetaTh - margin] ∪ [2π - thetaTh + margin, 2π - margin]
                const thetaNorm = theta % (2 * Math.PI);

                // 计算到两个安全边界的距离
                let dist1 = Math.abs(thetaNorm - bound1);
                let dist2 = Math.abs(thetaNorm - bound2);
                // 考虑跨周期
                dist1 = Math.min(dist1, Math.abs(thetaNorm - (bound1 + 2 * Math.PI)));
                dist2 = Math.min(dist2, Math.abs(thetaNorm - (bound2 - 2 * Math.PI)));

                if (dist1 <= dist2) {
                    hexGenerators[i] = bound1;
                } else {
                    hexGenerators[i] = bound2;
                }
            }
        }

        function angleFromScreen(clientX, clientY, idx) {
            const rect = renderer.domElement.getBoundingClientRect();
            const mx = ((clientX - rect.left) / rect.width) * 2 - 1;
            const my = -((clientY - rect.top) / rect.height) * 2 + 1;
            const ray = new THREE.Raycaster();
            ray.setFromCamera(new THREE.Vector2(mx, my), camera);
            // 统一方法：射线与截平面求交，然后从交点反推母线方位角
            let planePoint;
            if (currentParams.isEllipse || currentParams.isHyperbola) {
                planePoint = currentParams.center;
            } else if (currentParams.isParabola) {
                planePoint = currentParams.vertex;
            } else if (currentParams.isDegenerate) {
                planePoint = new THREE.Vector3(0, currentParams.coneHeight, 0);
            }
            if (!planePoint) return hexGenerators[idx];
            const plane = new THREE.Plane().setFromNormalAndCoplanarPoint(currentParams.planeNormal, planePoint);
            const target = new THREE.Vector3();
            ray.ray.intersectPlane(plane, target);
            if (!target) return hexGenerators[idx];
            // 从交点计算母线方位角：atan2(z, x) 相对于锥顶
            const apex = new THREE.Vector3(0, currentParams.coneHeight, 0);
            const dx = target.x - apex.x;
            const dz = target.z - apex.z;
            let theta = Math.atan2(dz, dx);
            // 保持点在原来的"半区"（避免跳转到对面的母线）
            const currentTheta = hexGenerators[idx];
            const diff = theta - currentTheta;
            if (diff > Math.PI) theta -= 2 * Math.PI;
            else if (diff < -Math.PI) theta += 2 * Math.PI;
            return theta;
        }

        function computeDandelin(alphaRad, gammaRad) {
            const H = CONE_H;
            const h0 = INTERCEPT;
            const sinA = Math.sin(alphaRad);
            const cosA = Math.cos(alphaRad);
            const betaRad = Math.PI / 2 - gammaRad;
            const sinB = Math.sin(betaRad);
            const cosB = Math.cos(betaRad);
            const nUnit = new THREE.Vector3(sinB, -cosB, 0).normalize();
            const P0 = new THREE.Vector3(0, h0, 0);

            const isEllipse = gammaRad > alphaRad + 1e-6;
            const isParabola = Math.abs(gammaRad - alphaRad) <= 1e-6;
            const isHyperbola = gammaRad < alphaRad - 1e-6;
            const isDegenerate = gammaRad < 1e-3;

            const y1 = (H * sinA + h0 * cosB) / (sinA + cosB);
            const R1 = (y1 - h0) * cosB;
            const O1 = new THREE.Vector3(0, y1, 0);
            const d1 = (O1.x - P0.x) * nUnit.x + (O1.y - P0.y) * nUnit.y + (O1.z - P0.z) * nUnit.z;
            const F1 = O1.clone().sub(nUnit.clone().multiplyScalar(d1));
            const y_t1 = H * sinA * sinA + y1 * cosA * cosA;
            const r1_circle = (H - y_t1) * Math.tan(alphaRad);

            let y2, R2, O2, F2, y_t2, r2_circle;
            let lowerValid = true;
            if (isParabola) {
                lowerValid = false;
                y2 = 0; R2 = 0; O2 = new THREE.Vector3(0,0,0); F2 = new THREE.Vector3(0,0,0);
                y_t2 = 0; r2_circle = 0;
            } else if (isHyperbola) {
                const denom = cosB - sinA;
                y2 = (h0 * cosB - H * sinA) / denom;
                R2 = (y2 - H) * sinA;
                if (R2 < 0.01 || y2 <= H) { lowerValid = false; y2 = H; R2 = 0; }
                O2 = new THREE.Vector3(0, y2, 0);
                const d2 = (O2.x - P0.x) * nUnit.x + (O2.y - P0.y) * nUnit.y + (O2.z - P0.z) * nUnit.z;
                F2 = O2.clone().sub(nUnit.clone().multiplyScalar(d2));
                y_t2 = H * sinA * sinA + y2 * cosA * cosA;
                r2_circle = (y_t2 - H) * Math.tan(alphaRad);
            } else {
                const denom = sinA - cosB;
                y2 = (H * sinA - h0 * cosB) / denom;
                R2 = (h0 - y2) * cosB;
                if (R2 <= 0.01) { lowerValid = false; y2 = 0; R2 = 0; }
                O2 = new THREE.Vector3(0, y2, 0);
                const d2 = (O2.x - P0.x) * nUnit.x + (O2.y - P0.y) * nUnit.y + (O2.z - P0.z) * nUnit.z;
                F2 = O2.clone().sub(nUnit.clone().multiplyScalar(d2));
                y_t2 = y2 + R2 * sinA;
                r2_circle = (H - y_t2) * Math.tan(alphaRad);
            }

            let center, aLen, bLen, uDir, vDir;
            let directrixPoint = null, directrixDir = null, vertex = null, focalParam = null;
            let isCircle = false;
            if (isEllipse) {
                const F1F2 = new THREE.Vector3().subVectors(F2, F1);
                const c = F1F2.length() / 2;
                isCircle = c < 1e-6;
                const a = Math.abs(y_t1 - y_t2) / (2 * cosA);
                const b = Math.sqrt(Math.max(0, a * a - c * c));
                center = new THREE.Vector3().addVectors(F1, F2).multiplyScalar(0.5);
                if (F1F2.length() < 1e-6) {
                    uDir = new THREE.Vector3(1, 0, 0);
                } else {
                    uDir = F1F2.clone().normalize();
                }
                vDir = new THREE.Vector3().crossVectors(nUnit, uDir).normalize();
                aLen = a; bLen = b;
            } else if (isHyperbola) {
                const F1F2 = new THREE.Vector3().subVectors(F1, F2);
                const c = F1F2.length() / 2;
                const a = Math.abs(y_t1 - y_t2) / (2 * cosA);
                const b = Math.sqrt(Math.max(0, c * c - a * a));
                center = new THREE.Vector3().addVectors(F1, F2).multiplyScalar(0.5);
                uDir = F1F2.clone().normalize();
                vDir = new THREE.Vector3().crossVectors(nUnit, uDir).normalize();
                aLen = a; bLen = b;
            } else {
                directrixDir = new THREE.Vector3().crossVectors(nUnit, new THREE.Vector3(0,1,0)).normalize();
                if (directrixDir.length() < 0.001) directrixDir = new THREE.Vector3(1,0,0);
                const xL = cosB * (y_t1 - h0) / sinB;
                directrixPoint = new THREE.Vector3(xL, y_t1, 0);
                uDir = directrixDir.clone();
                vDir = new THREE.Vector3().crossVectors(uDir, nUnit).normalize();
                const toFocus = new THREE.Vector3().subVectors(F1, directrixPoint);
                const proj = toFocus.dot(vDir);
                if (proj < 0) vDir.negate();
                focalParam = Math.abs(toFocus.dot(vDir));
                vertex = F1.clone().addScaledVector(vDir, -focalParam/2);
                center = vertex.clone();
                aLen = 0; bLen = 0;
            }

            const R_cone = getConeRadius(alphaRad);

            function computeDirectrixForLevel(y_t, n, h0_val) {
                const dir = new THREE.Vector3().crossVectors(n, new THREE.Vector3(0, 1, 0)).normalize();
                if (dir.length() < 0.001) dir.set(1, 0, 0);
                let pt;
                if (Math.abs(n.z) > 0.001) {
                    const z = -n.y * (y_t - h0_val) / n.z;
                    pt = new THREE.Vector3(0, y_t, z);
                } else if (Math.abs(n.x) > 0.001) {
                    const x = -n.y * (y_t - h0_val) / n.x;
                    pt = new THREE.Vector3(x, y_t, 0);
                } else {
                    pt = new THREE.Vector3(0, y_t, 0);
                }
                return { point: pt, dir: dir };
            }

            let directrixData = [];
            if (isEllipse || isHyperbola) {
                const d1 = computeDirectrixForLevel(y_t1, nUnit, h0);

                    directrixData.push({ point: d2.point, dir: d2.dir, label: 'K₂' });

                directrixData.push({ point: d.point, dir: d.dir, label: 'K' });
                directrixPoint = d.point;
                directrixDir = d.dir;
            }

            return {
                center, aLen, bLen, uDir, vDir,
                O1, R1, y_t1, r1: r1_circle, F1,

                directrixPoint, directrixDir, vertex, focalParam,
                directrixData,
                isDegenerate,
                isCircle,
                alphaRad
            };
        }

        function getConicPoint(params, t, branch = 0) {
            if (params.isDegenerate) {
                const sign = (branch >= 0) ? 1 : -1;
                const cosA = Math.cos(params.alphaRad);
                const sinA = Math.sin(params.alphaRad);
                const dir = new THREE.Vector3(0, -cosA, sign * sinA);
                const apex = new THREE.Vector3(0, params.coneHeight, 0);
                return apex.clone().addScaledVector(dir, t);
            }
            if (params.isEllipse) {
                const cos_t = Math.cos(t), sin_t = Math.sin(t);
                return new THREE.Vector3(
                    params.center.x + params.aLen * cos_t * params.uDir.x + params.bLen * sin_t * params.vDir.x,
                    params.center.y + params.aLen * cos_t * params.uDir.y + params.bLen * sin_t * params.vDir.y,
                    params.center.z + params.aLen * cos_t * params.uDir.z + params.bLen * sin_t * params.vDir.z
                );
            } else if (params.isHyperbola) {
                const cosh_t = Math.cosh(t), sinh_t = Math.sinh(t);
                const sign = (branch >= 0) ? 1 : -1;
                return new THREE.Vector3(
                    params.center.x + sign * params.aLen * cosh_t * params.uDir.x + params.bLen * sinh_t * params.vDir.x,
                    params.center.y + sign * params.aLen * cosh_t * params.uDir.y + params.bLen * sinh_t * params.vDir.y,
                    params.center.z + sign * params.aLen * cosh_t * params.uDir.z + params.bLen * sinh_t * params.vDir.z
                );
            } else if (params.isParabola) {
                const p = Math.max(params.focalParam, 0.1);
                return params.vertex.clone()
                    .addScaledVector(params.uDir, t)
                    .addScaledVector(params.vDir, (t * t) / (2 * p));
            }
            return new THREE.Vector3();
        }

        function getConicPointFromGenerator(params, theta) {
            const apex = new THREE.Vector3(0, params.coneHeight, 0);
            const sinA = Math.sin(params.alphaRad);
            const cosA = Math.cos(params.alphaRad);
            const dir = new THREE.Vector3(
                Math.cos(theta) * sinA,
                -cosA,
                Math.sin(theta) * sinA

            const s = P0.clone().sub(apex).dot(params.planeNormal) / dirDotN;
            return apex.clone().addScaledVector(dir, s);
        }

        function adjustHexGeneratorsToRedBranch(params) {
            if (!params.isHyperbola) return;
            const H = params.coneHeight;
            const h0 = INTERCEPT;
            const sinA = Math.sin(params.alphaRad);
            const cosA = Math.cos(params.alphaRad);
            const ny = params.planeNormal.y;
            const nx = params.planeNormal.x;
            const cosG = nx;
            const sinG_val = -ny;

            const threshold = -cosA * sinG_val / (sinA * cosG);
            const clampedTh = Math.max(-1, Math.min(1, threshold));
            const thetaTh = Math.acos(clampedTh);
            const margin = 3 * Math.PI / 180;

            const bound1 = thetaTh - margin;
            const bound2 = 2 * Math.PI - thetaTh + margin;

            for (let i = 0; i < 6; i++) {
                let theta = hexGenerators[i];
                const dir = new THREE.Vector3(
                    Math.cos(theta) * sinA,
                    -cosA,
                    Math.sin(theta) * sinA
                );
                const P0 = new THREE.Vector3(0, h0, 0);
                const apex = new THREE.Vector3(0, H, 0);
                const dirDotN = dir.dot(params.planeNormal);
                if (Math.abs(dirDotN) < 1e-10) continue;
                const s = P0.clone().sub(apex).dot(params.planeNormal) / dirDotN;

                if (s > 0) continue;

                const thetaNorm = theta % (2 * Math.PI);

                let dist1 = Math.abs(thetaNorm - bound1);
                let dist2 = Math.abs(thetaNorm - bound2);
                dist1 = Math.min(dist1, Math.abs(thetaNorm - (bound1 + 2 * Math.PI)));
                dist2 = Math.min(dist2, Math.abs(thetaNorm - (bound2 - 2 * Math.PI)));

                if (dist1 <= dist2) {
                    hexGenerators[i] = bound1;
                } else {
                    hexGenerators[i] = bound2;
                }
            }
        }

            return { A, B, dir };
        }

                const sinA = Math.sin(params.alphaRad);

                    pts1.push(getConicPoint(params, t, 1));

        function updateInterLine() {
            if (!interLine) return;
            const pts = [];
            if (interAB && interAB.visible) pts.push(interAB.position.clone());
            if (interBC && interBC.visible) pts.push(interBC.position.clone());
            if (interCA && interCA.visible) pts.push(interCA.position.clone());
            if (pts.length >= 2) {
                let p1, p2;
                if (pts.length === 3) {
                    let maxDist = 0, i1 = 0, i2 = 1;
                    for (let i = 0; i < 3; i++) {
                        for (let j = i + 1; j < 3; j++) {
                            const d = pts[i].distanceTo(pts[j]);
                            if (d > maxDist) { maxDist = d; i1 = i; i2 = j; }
                        }
                    }
                    p1 = pts[i1]; p2 = pts[i2];
                } else {
                    p1 = pts[0]; p2 = pts[1];
                }
                const dir = new THREE.Vector3().subVectors(p2, p1).normalize();
                const ext = 15;
                const start = p1.clone().sub(dir.clone().multiplyScalar(ext));
                const end = p2.clone().add(dir.clone().multiplyScalar(ext));
                if (interLine.geometry) interLine.geometry.dispose();
                interLine.geometry = new THREE.BufferGeometry().setFromPoints([start, end]);
                interLine.computeLineDistances();
                interLine.visible = hexVisible && currentParams && !currentParams.isDegenerate;
            } else {
                interLine.visible = false;
            }
            if (labelP1) { labelP1.position.copy(interAB.position); labelP1.visible = interAB.visible; }
            if (labelP2) { labelP2.position.copy(interBC.position); labelP2.visible = interBC.visible; }
            if (labelP3) { labelP3.position.copy(interCA.position); labelP3.visible = interCA.visible; }
        }

        function setTri(mesh, p1, p2, p3) {
            if (!mesh) return;
            if (mesh.geometry) mesh.geometry.dispose();
            mesh.geometry = new THREE.BufferGeometry();
            mesh.geometry.setAttribute('position', new THREE.BufferAttribute(new Float32Array([p1.x, p1.y, p1.z, p2.x, p2.y, p2.z, p3.x, p3.y, p3.z]), 3));
            mesh.geometry.setIndex([0,1,2]); mesh.geometry.computeVertexNormals();
        }

        function updateScene(angle) {
            if (!currentParams) return;
            try {
                let t, P, branch = 0;
                if (currentParams.isDegenerate) {
                    const cosA = Math.cos(currentParams.alphaRad);
                    const sinA = Math.sin(currentParams.alphaRad);
                    const apex = new THREE.Vector3(0, currentParams.coneHeight, 0);
                    const range = currentParams.coneHeight / cosA;
                    if (angle <= 180) {
                        const frac = angle / 180;
                        const s = frac * (2 * range) - range;
                        if (s < 0) {
                            P = apex.clone().addScaledVector(new THREE.Vector3(0, -cosA, sinA), -s);
                        } else {
                            P = apex.clone().addScaledVector(new THREE.Vector3(0, cosA, -sinA), s);
                        }
                    } else {
                        const frac = (angle - 180) / 180;
                        const s = frac * (2 * range) - range;
                        if (s < 0) {
                            P = apex.clone().addScaledVector(new THREE.Vector3(0, -cosA, -sinA), -s);
                        } else {
                            P = apex.clone().addScaledVector(new THREE.Vector3(0, cosA, sinA), s);
                        }
                    }
                } else if (currentParams.isEllipse) {
                    t = angle * Math.PI / 180;
                    P = getConicPoint(currentParams, t);
                } else if (currentParams.isHyperbola) {
                    const range = 2.5;
                    if (angle <= 180) {
                        branch = -1;
                        t = -range + (angle / 180) * range * 2;
                    } else {
                        branch = 1;
                        t = -range + ((angle - 180) / 180) * range * 2;
                    }
                    P = getConicPoint(currentParams, t, branch);
                } else {
                    t = (angle - 180) * 0.05;
                    P = getConicPoint(currentParams, t);
                }
                pMesh.position.copy(P); labelP.position.copy(P);
                if (currentParams.isDegenerate) {
                    aMesh.position.copy(P); bMesh.position.copy(P);
                    labelT1.position.copy(P); labelT2.position.copy(P);
                    document.getElementById('p-value').textContent = angle.toFixed(0) + '°';
                    document.getElementById('val-pf1').textContent = '--';
                    document.getElementById('val-pt1').textContent = '--';
                    document.getElementById('val-pf2').textContent = '--';
                    document.getElementById('val-pt2').textContent = '--';
                    document.getElementById('val-tt').textContent = '--';
                    document.getElementById('info-sum').innerHTML = '退化：两条相交直线 <span class="info-const">母线</span>';
                    const apex = new THREE.Vector3(0, currentParams.coneHeight, 0);
                    const genDir = new THREE.Vector3().subVectors(P, apex);
                    const genLen = genDir.length();
                    if (genLen > 1e-6) {
                        genDir.multiplyScalar(40 / genLen);
                        setLine(lineGenerator, apex, apex.clone().add(genDir));
                    }
                    return;
                }
                const { A, B, dir } = getGeneratorIntersections(currentParams, P);
                aMesh.position.copy(A); bMesh.position.copy(B);
                labelT1.position.copy(A); labelT2.position.copy(B);

                setLine(lineAP, A, P); if (currentParams.lowerValid) setLine(lineBP, B, P);
                setLine(linePF1, P, currentParams.F1); if (currentParams.lowerValid) setLine(linePF2, P, currentParams.F2);
                setLine(lineAB, A, B);
                const apex = new THREE.Vector3(0, currentParams.coneHeight, 0);
                setLine(lineGenerator, apex, apex.clone().add(dir.clone().multiplyScalar(40)));

                setLine(lineO1P, currentParams.O1, P); setLine(lineO1A, currentParams.O1, A); setLine(lineO1F1, currentParams.O1, currentParams.F1);
                if (currentParams.lowerValid) {
                    setLine(lineO2P, currentParams.O2, P); setLine(lineO2B, currentParams.O2, B); setLine(lineO2F2, currentParams.O2, currentParams.F2);
                }

                if (currentParams.isParabola) {
                    if (pointK && linePK && labelK && _directrixData && _directrixData.lines.length > 0) {
                        const d = _directrixData.lines[0];
                        const L0 = d.point, dirL = d.dir;
                        const K = L0.clone().addScaledVector(dirL, P.clone().sub(L0).dot(dirL));
                        pointK.position.copy(K); labelK.position.copy(K);
                        setLine(linePK, P, K);
                        linePK.computeLineDistances();
                    }
                    if (pointH && labelH) {
                        const H = new THREE.Vector3(P.x, currentParams.y_t1, P.z);
                        pointH.position.copy(H); labelH.position.copy(H);
                    }
                    if (linePH && pointH) {
                        const H = pointH.position;
                        setLine(linePH, P, H);
                    }
                    if (pointK && lineHK && pointH && _directrixData && _directrixData.lines.length > 0) {
                        const d = _directrixData.lines[0];
                        const L0 = d.point, dirL = d.dir;
                        const K = L0.clone().addScaledVector(dirL, P.clone().sub(L0).dot(dirL));
                        setLine(lineHK, K, pointH.position);
                    }
                    if (lineHT1 && pointH) {
                        setLine(lineHT1, A, pointH.position);
                    }
                    if (pointK && pointH && _directrixData && _directrixData.lines.length > 0) {
                        const d = _directrixData.lines[0];
                        const L0 = d.point, dirL = d.dir;
                        const K = L0.clone().addScaledVector(dirL, P.clone().sub(L0).dot(dirL));
                        setTri(triKHP, K, pointH.position, P);
                        setTri(triT1HP, A, pointH.position, P);
                    }
                } else if (currentParams.isEllipse || currentParams.isHyperbola) {
                    if (_directrixData && _directrixData.lines.length >= 1) {
                        const d1 = _directrixData.lines[0];
                        const L0_1 = d1.point, dirL_1 = d1.dir;

                        if (pointK1) pointK1.position.copy(K1);
                        if (labelK1) labelK1.position.copy(K1);

                }
                document.getElementById('p-value').textContent = angle.toFixed(0) + '°';

                if (draggingHexIdx >= 0 && hexPoints.length === 6) {
                    const Pt = getConicPointFromGenerator(currentParams, hexGenerators[draggingHexIdx]);
                    if (Pt) { hexPoints[draggingHexIdx].position.copy(Pt); hexLabels[draggingHexIdx].position.copy(Pt); }
                }

                    const a = hexPoints[3].position, b = hexPoints[4].position, c = hexPoints[5].position;
                    updateHexLine(lineAb, A, b); updateHexLine(lineBa, B, a);
                    updateHexLine(lineBc, B, c); updateHexLine(lineCb, C, b);
                    updateHexLine(lineCa, C, a); updateHexLine(lineAc, A, c);
                    function lineIntersect(p1, p2, p3, p4) {
                        const d1 = new THREE.Vector3().subVectors(p2, p1);
                        const d2 = new THREE.Vector3().subVectors(p4, p3);
                        const diff = new THREE.Vector3().subVectors(p3, p1);
                        const crossD = new THREE.Vector3().crossVectors(d1, d2);
                        const crossDiff = new THREE.Vector3().crossVectors(diff, d2);
                        const lenSq = crossD.lengthSq();
                        if (lenSq < 1e-8) return null;
                        const t = crossDiff.dot(crossD) / lenSq;
                        return p1.clone().addScaledVector(d1, t);
                    }
                    const iAB = lineIntersect(A, b, B, a);
                    const iBC = lineIntersect(B, c, C, b);
                    const iCA = lineIntersect(C, a, A, c);
                    if (iAB) { interAB.position.copy(iAB); interAB.visible = true; } else { interAB.visible = false; }
                    if (iBC) { interBC.position.copy(iBC); interBC.visible = true; } else { interBC.visible = false; }
                    if (iCA) { interCA.position.copy(iCA); interCA.visible = true; } else { interCA.visible = false; }
                    updateInterLine();
                }

                const distPT1 = P.distanceTo(A);
                const distPT2 = currentParams.lowerValid ? P.distanceTo(B) : 0;
                const distT1T2 = currentParams.lowerValid ? A.distanceTo(B) : 0;
                const hidePT1 = hexVisible && (currentParams.isEllipse || currentParams.isHyperbola || currentParams.isParabola);

                document.getElementById('val-tt').textContent = currentParams.lowerValid ? distT1T2.toFixed(3) : '--';
                const infoSum = document.getElementById('info-sum');
                if (currentParams.isEllipse) {
                    const sum = distPF1 + distPF2;
                    infoSum.innerHTML = 'PF₁ + PF₂ = <span class="info-val" id="val-diff">' + sum.toFixed(3) + '</span> <span class="info-const">定值</span>';
                } else if (currentParams.isHyperbola) {
                    const diff = Math.abs(distPF2 - distPF1);
                    infoSum.innerHTML = '|PF₂ − PF₁| = <span class="info-val" id="val-diff">' + diff.toFixed(3) + '</span> <span class="info-const">定值</span>';
                } else if (currentParams.isParabola) {
                    infoSum.innerHTML = 'PF = PK = <span class="info-val" id="val-diff">' + distPF1.toFixed(3) + '</span> <span class="info-const">定值</span>';
                }

                const time = Date.now() * 0.0015;
                const pulse = (Math.sin(time) + 1) * 0.5;
                const baseOp = 0.30;
                const pulseOp = 0.35;
                if (triAO1P && triF1O1P) {

                    const blueOp = baseOp + (1 - pulse) * pulseOp;
                    triBO2P.material.opacity = blueOp;
                    triF2O2P.material.opacity = blueOp;
                    triBO2P.material.color.setHex(0x3333ff);
                    triF2O2P.material.color.setHex(0x3333ff);
                }
            } catch (e) {
                console.error('updateScene error:', e);
            }

        function findHexAt(clientX, clientY) {
            if (!hexVisible) return -1;
            for (let i = 0; i < 6; i++) {
                const sp = getHexScreenPos(i);
                if (!sp) continue;
                const dx = clientX - sp.x, dy = clientY - sp.y;
                const dist = Math.sqrt(dx * dx + dy * dy);
                const r = HEX_HIT_R * 80;
                if (dist < r) return i;
            }
            return -1;
        }

        function angleFromScreen(clientX, clientY, idx) {
