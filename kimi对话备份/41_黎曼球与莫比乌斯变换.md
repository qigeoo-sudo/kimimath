function computeAndDisplayCR() {
  if (crFixedPoints.length < 4) return;

  const a = complexParams.a, b = complexParams.b;
  const c = complexParams.c, d = complexParams.d;

  // 获取4个点的三种坐标
  const planeZs = [];
  const stereoZs = []; // 通过逆球极投影从球面回到平面
  const mobiusZs = [];

  for (let i = 0; i < 4; i++) {
    const fixed = crFixedPoints[i];
    const posArr = planeOrigCurveLines[fixed.branchIdx].geometry.attributes.position.array;
    const count = planeOrigCurveLines[fixed.branchIdx].geometry.drawRange.count;
    const safeCount = isFinite(count) ? count : 0;
    if (fixed.pointIdx >= safeCount) return;
    
    const z = { re: posArr[fixed.pointIdx * 3], im: posArr[fixed.pointIdx * 3 + 2] };
    planeZs.push(z);
    
    // 球极：先投影到球面，再逆投影回平面（得到相同的z，但为了代码一致性）
    const pProj = stereoProject(z);
    const zStereo = stereoInverse(pProj);
    stereoZs.push(zStereo || z);
    
    // 莫比
    const w = mobiusTransform(z, a, b, c, d);
    mobiusZs.push(w || z);
  }

  const crStereo = computeCR(stereoZs[0], stereoZs[1], stereoZs[2], stereoZs[3]);
  const crMobius = computeCR(mobiusZs[0], mobiusZs[1], mobiusZs[2], mobiusZs[3]);

  const curveColor = curveColors[currentCurve] || '#333333';
  
  const crPlaneEl = document.getElementById('cr-plane');

    crStereoEl.textContent = formatCR(crStereo);
    crStereoEl.style.color = curveColor;
  }
  if (crMobiusEl) {
    crMobiusEl.textContent = formatCR(crMobius);

function computeCR(z1, z2, z3, z4) {
  const num = cMul(cSub(z1, z3), cSub(z2, z4));
  const den = cMul(cSub(z1, z4), cSub(z2, z3));

### 核心原因：交比是 Möbius 变换的**普适不变量**

交比 (z1,z2;z3,z4) 对 **任意四个复数点**（只要分母不为零）都是 Möbius 变换的不变量。它**不依赖**于这四点是否共圆、共线、共双曲线，或者任何其他代数曲线。**证明非常简单：**
w=cz+daz+b 直接计算可得：
wk−wlwi−wj=(zk−zl)(czi+d)(czj+d)(zi−zj)(czk+d)(czl+d) 当构造交比时，这些 (cz+d) 因子会**完全抵消**：
(w1,w2;w3,w4)=(w1−w4)(w2−w3)(w1−w3)(w2−w4)=(z1−z4)(z2−z3)(z1−z3)(z2−z4)=(z1,z2;z3,z4) **没有任何步骤要求四点共圆。**

### 澄清一个可能的误解

> "交比只保四点共圆"

这句话如果存在，通常是在说**另一个相关但不同的性质**：

- Möbius 变换把**圆和直线**映射为**圆和直线**
- 因此，如果四点原本共圆，变换后仍然共圆
- 但这只是 Möbius 变换的**几何保持性**，不是交比不变性的**前提条件**

交比不变性是**代数恒等式**，对任意四点都成立。四点可以：

- 在双曲线上 ✅
- 在抛物线上 ✅
- 在椭圆曲线上 ✅
- 在任意散乱位置上 ✅

只要分母不为零，交比就不变。

### 为什么球极投影的交比也相同？

球极投影（从北极到平面）本身也是一个**Möbius 变换的特例**（可以写成 w=1−zx+iy 的形式，其中 (x,y,z) 是单位球面上的点）。因此，球极投影也保持交比不变。在代码中，球极的四个点直接拷贝了平面数值，所以交比自然相同。

### 所以你的观察说明了什么？

你的代码**正确实现**了数学事实：

- **平面** = **球极** = **莫比**（理论上完全相同）
- 你看到的微小差异（如果有的话）只是浮点精度误差

如果你希望看到**不同的**交比数值，那需要：

- 对球极使用**非保角**的投影（比如随便一个非 Möbius 映射）
- 或者对莫比使用**非 Möbius** 的变换（比如 w=z2 或 w=zˉ ）

但既然你做的是**黎曼球 + Möbius 变换**，三者交比相同正是这个数学结构的核心特征。
